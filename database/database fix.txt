CREATE TABLE guru (
    nip_g VARCHAR(20) PRIMARY KEY,
    kd_guru VARCHAR(10) NOT NULL UNIQUE,
    nama VARCHAR(50) NOT NULL,
    password VARCHAR(12) NOT NULL,
    tgl_lahir DATE,
    alamat TEXT,
    no_telp VARCHAR(13) NOT NULL,
    jkel ENUM('Laki - laki','Perempuan') DEFAULT 'Laki - laki',
    keahlian VARCHAR(25) NOT NULL
);

CREATE TABLE tbl_kelas (
    id_kelas VARCHAR(10) PRIMARY KEY NOT NULL,
    kelas VARCHAR(20) NOT NULL,
    kd_guru VARCHAR(10) NOT NULL,
    FOREIGN KEY (`kd_guru`) REFERENCES `guru`(`kd_guru`) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tbl_jadwal (
    id_mapel VARCHAR(10) PRIMARY KEY NOT NULL,
    nama_mapel VARCHAR(15) NOT NULL,
    kd_guru VARCHAR(10) NOT NULL,
    jam TIME,
    kelas VARCHAR(20) NOT NULL,
    FOREIGN KEY (`kd_guru`) REFERENCES `guru`(`kd_guru`) ON DELETE RESTRICT ON UPDATE CASCADE,
    FOREIGN KEY (`kelas`) REFERENCES `tbl_kelas`(`id_kelas`) ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tbl_siswa (
    nisn VARCHAR(15) PRIMARY KEY,
    nama VARCHAR(30) NOT NULL,
    tgl_lahir DATE,
    alamat TEXT,
    notelp VARCHAR(15) NOT NULL,
    jkel ENUM('Laki - laki','Perempuan') DEFAULT 'Laki - laki',
    angkatan VARCHAR(4) NOT NULL,
    nama_wali VARCHAR(30) NOT NULL,
    no_wali VARCHAR(15) NOT NULL,
    jurusan VARCHAR(25) NOT NULL,
    kelas VARCHAR(20) NOT NULL,
    FOREIGN KEY (`kelas`) REFERENCES `tbl_kelas`(`id_kelas`) ON UPDATE RESTRICT ON DELETE CASCADE
);

CREATE TABLE tbl_absen (
    nisn VARCHAR(15) PRIMARY KEY,
    kelas VARCHAR(20) NOT NULL,
    kehadiran INT(3),
    jumlah INT(3),
    FOREIGN KEY (`kelas`) REFERENCES `tbl_kelas`(`id_kelas`) ON UPDATE RESTRICT ON DELETE CASCADE
);

CREATE TABLE tbl_report (
    id_report VARCHAR(15) PRIMARY KEY,
    jumlah INT(3),
    nama VARCHAR(30),
    kelas VARCHAR(20),
    FOREIGN KEY (`nama`) REFERENCES `tbl_siswa`(`nisn`) ON UPDATE RESTRICT ON DELETE CASCADE,
    FOREIGN KEY (`kelas`) REFERENCES `tbl_kelas`(`id_kelas`) ON UPDATE RESTRICT ON DELETE CASCADE
);

CREATE TABLE tbl_transaksi (
    no_transaksi VARCHAR(10) PRIMARY KEY,
    tgl DATE,
    nisn VARCHAR(15) NOT NULL,
    FOREIGN KEY (`nisn`) REFERENCES `tbl_siswa`(`nisn`) ON UPDATE RESTRICT ON DELETE CASCADE
);

CREATE TABLE 'users' (
  `nip` int(20) NOT NULL,
  `nama` varchar(25) NOT NULL,
  `password` varchar(20) NOT NULL,
  `lvl` varchar(15) NOT NULL,
  `status` varchar(2) NOT NULL
);

CREATE TABLE tbl_nilai (
    id_nilai INT AUTO_INCREMENT PRIMARY KEY,
    nisn VARCHAR(15) NOT NULL,            
    id_mapel VARCHAR(10) NOT NULL,        
    kd_guru VARCHAR(10) NOT NULL,         
    jenis_nilai VARCHAR(20) NOT NULL,     
    skor INT NOT NULL,                    
    FOREIGN KEY (nisn) REFERENCES tbl_siswa(nisn) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (id_mapel) REFERENCES tbl_jadwal(id_mapel) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (kd_guru) REFERENCES guru(kd_guru) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE tbl_absen_guru (
    id_absen INT PRIMARY KEY AUTO_INCREMENT,
    kd_guru VARCHAR(10), 
    tgl DATE,
    kehadiran INT(1) DEFAULT 0,
    keterangan VARCHAR(50),
    -- Tambahkan Foreign Key langsung di sini
    CONSTRAINT fk_absen_guru FOREIGN KEY (kd_guru) 
    REFERENCES guru(kd_guru) 
    ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE tbl_absen_user (
    id_absen INT PRIMARY KEY AUTO_INCREMENT,
    nip VARCHAR(10), 
    tgl DATE,
    kehadiran INT(1) DEFAULT 0,
    keterangan VARCHAR(50),
    -- Tambahkan Foreign Key langsung di sini
    CONSTRAINT fk_absen_user FOREIGN KEY (nip) 
    REFERENCES users(nip) 
    ON DELETE CASCADE ON UPDATE CASCADE
);
